/******************************************************* * PERSENTASE KEHADIRAN * SMP ISLAM CIPAKU * * File ini DITAMBAHKAN sebagai file Apps Script baru. * Tidak mengganti rekap_router.gs dan tidak menyentuh jalur scan. * * Sumber data: * - SISWA * - JADWAL * - PRESENSI * * Rumus: * Persentase = (Hadir + Terlambat) / kesempatan yang dihitung * 100 * * HAID tidak dimasukkan ke penyebut, sehingga tidak menurunkan persentase. * Sesi hari ini baru dihitung setelah jam TUTUP lewat. *******************************************************/ /* ===================================================== UTIL ===================================================== */ function persenBulat2_(angka) { if ( !isFinite(angka) ) return 0; return Math.round( angka * 100 ) / 100; } function persenNilai_(hadir, kesempatan) { if ( !kesempatan ) return 0; return persenBulat2_( (hadir / kesempatan) * 100 ); } function persenTanggalIso_(nilai) { if ( Object.prototype.toString.call(nilai) === "[object Date]" && !isNaN(nilai.getTime()) ) { return Utilities.formatDate( nilai, TIMEZONE, "yyyy-MM-dd" ); } const teks = String( nilai == null ? "" : nilai ).trim(); if (!teks) return ""; let m = teks.match( /^(\d{4})[-\/](\d{1,2})[-\/](\d{1,2})$/ ); if (m) { return ( m[1] + "-" + String(Number(m[2])).padStart(2, "0") + "-" + String(Number(m[3])).padStart(2, "0") ); } m = teks.match( /^(\d{1,2})[-\/](\d{1,2})[-\/](\d{4})$/ ); if (m) { return ( m[3] + "-" + String(Number(m[2])).padStart(2, "0") + "-" + String(Number(m[1])).padStart(2, "0") ); } return ""; } function persenDateDariIso_(iso) { const p = String( iso || "" ).split("-"); if ( p.length !== 3 ) return null; const d = new Date( Number(p[0]), Number(p[1]) - 1, Number(p[2]), 12, 0, 0, 0 ); return isNaN( d.getTime() ) ? null : d; } function persenHariIndonesia_(iso) { const d = persenDateDariIso_( iso ); if (!d) return ""; const en = Utilities.formatDate( d, TIMEZONE, "EEEE" ); const map = { Monday: "Senin", Tuesday: "Selasa", Wednesday: "Rabu", Thursday: "Kamis", Friday: "Jumat", Saturday: "Sabtu", Sunday: "Minggu" }; return map[en] || ""; } function persenTambahHari_(iso, jumlah) { const d = persenDateDariIso_( iso ); if (!d) return ""; d.setDate( d.getDate() + jumlah ); return Utilities.formatDate( d, TIMEZONE, "yyyy-MM-dd" ); } function persenSelisihHari_(mulai, akhir) { const a = persenDateDariIso_( mulai ); const b = persenDateDariIso_( akhir ); if ( !a || !b ) return -1; return Math.floor( (b.getTime() - a.getTime()) / 86400000 ); } function persenJamMenit_(nilai) { const teks = String( nilai == null ? "" : nilai ) .trim() .toUpperCase() .replace(/\s+/g, "_"); } function persenHeaderMap_(barisHeader) { const map = {}; for ( let i = 0; i < barisHeader.length; i++ ) { const key = persenNormalisasiHeader_( barisHeader[i] ); if (key) { map[key] = i; } } return map; } function persenAmbilKolom_(map, daftarNama, fallback) { for ( let i = 0; i < daftarNama.length; i++ ) { const k = persenNormalisasiHeader_( daftarNama[i] ); if ( Object.prototype.hasOwnProperty.call( map, k ) ) { return map[k]; } } return fallback; } /* ===================================================== JADWAL UNTUK TANGGAL TERTENTU ===================================================== */ function persenJadwalTanggal_( isoTanggal, dataJadwal ) { const hari = persenHariIndonesia_( isoTanggal ); const rutin = []; const khusus = []; const tambahan = []; for ( let i = 1; i < dataJadwal.length; i++ ) { const row = dataJadwal[i]; const hariRow = String( row[0] || "" ).trim(); const kegiatan = String( row[1] || "" ).trim(); if (!kegiatan) continue; const jenisRaw = String( row[4] || "RUTIN" ) .trim() .toUpperCase(); if ( jenisRaw === "NONAKTIF" ) { continue; } const jenis = ( jenisRaw === "KHUSUS" || jenisRaw === "TAMBAHAN" ) ? jenisRaw : "RUTIN"; const mulai = String( row[2] == null ? "" : row[2] ).trim(); const tutup = String( row[3] == null ? "" : row[3] ).trim(); const tanggalMulai = persenTanggalIso_( row[5] ); const tanggalAkhir = persenTanggalIso_( row[6] ); /* * HARI kosong = berlaku setiap hari di dalam rentang tanggal. */ if ( hariRow && hariRow.toLowerCase() !== hari.toLowerCase() ) { continue; } if ( tanggalMulai && isoTanggal < tanggalMulai ) { continue; } if ( tanggalAkhir && isoTanggal > tanggalAkhir ) { continue; } /* * KHUSUS dan TAMBAHAN wajib memiliki minimal satu batas tanggal. */ if ( jenis !== "RUTIN" && !tanggalMulai && !tanggalAkhir ) { continue; } const item = { kegiatan: kegiatan, mulai: mulai, tutup: tutup, jenis: jenis }; if ( jenis === "KHUSUS" ) { khusus.push(item); } else if ( jenis === "TAMBAHAN" ) { tambahan.push(item); } else { rutin.push(item); } } const hasil = khusus.length > 0 ? khusus.concat(tambahan) : rutin.concat(tambahan); hasil.sort( function(a, b) { return ( persenJamMenit_(a.mulai) - persenJamMenit_(b.mulai) ); } ); return hasil; } /* ===================================================== SISWA ===================================================== */ function persenBacaSiswa_() { const sheet = getSS().getSheetByName( "SISWA" ); const cNis = persenAmbilKolom_( h, ["NIS", "NIS/NISN", "NISN"], 1 ); const cNama = persenAmbilKolom_( h, ["NAMA"], 2 ); const cKelas = persenAmbilKolom_( h, ["KELAS"], 3 ); const cJk = persenAmbilKolom_( h, ["JK", "JENIS_KELAMIN"], 4 ); const cStatus = persenAmbilKolom_( h, ["STATUS"], 5 ); const cMasuk = persenAmbilKolom_( h, ["TANGGAL_MASUK"], 7 ); const cNonaktif = persenAmbilKolom_( h, ["TANGGAL_NONAKTIF"], 8 ); const hasil = []; } return hasil; } function persenSiswaAktifTanggal_( siswa, isoTanggal ) { /* * Jika tanggal masuk tersedia, siswa baru dihitung sejak tanggal itu. */ if ( siswa.tanggalMasuk && isoTanggal < siswa.tanggalMasuk ) { return false; } /* * Jika tanggal nonaktif tersedia, setelah tanggal itu tidak dihitung lagi. * Pada tanggal nonaktifnya sendiri masih dianggap bagian periode aktif. */ if ( siswa.tanggalNonaktif && isoTanggal > siswa.tanggalNonaktif ) { return false; } /* * Untuk data lama yang belum punya tanggal masuk/nonaktif: * status AKTIF dipakai sebagai pengaman. */ if ( !siswa.tanggalMasuk && !siswa.tanggalNonaktif ) { return ( siswa.status === "AKTIF" ); } return true; } /* ===================================================== PRESENSI ===================================================== */ function persenBacaPresensiMap_( mulai, akhir ) { const sheet = getSS().getSheetByName( "PRESENSI" ); const cTanggal = persenAmbilKolom_( h, ["TANGGAL"], 1 ); const cId = persenAmbilKolom_( h, ["ID_SISWA", "ID"], 3 ); const cKegiatan = persenAmbilKolom_( h, ["KEGIATAN"], 7 ); const cStatus = persenAmbilKolom_( h, ["STATUS"], 8 ); const map = {}; for ( let i = 1; i < data.length; i++ ) { const row = data[i]; const tanggal = persenTanggalIso_( row[cTanggal] ); if ( !tanggal || tanggal < mulai || tanggal > akhir ) { continue; } const id = String( row[cId] || "" ).trim(); const kegiatan = String( row[cKegiatan] || "" ).trim(); if ( !id || !kegiatan ) continue; const status = String( row[cStatus] || "" ) .trim() .toUpperCase(); const key = tanggal + "|" + kegiatan.toLowerCase() + "|" + id.toUpperCase(); /* * Jika ada lebih dari satu baris karena koreksi/manual lama, * baris paling bawah dianggap yang terbaru. */ map[key] = status; } return map; } /* ===================================================== AGREGASI ===================================================== */ function persenObjekAwal_(label) { return { label: label, kesempatan: 0, dihitung: 0, hadir: 0, terlambat: 0, sakit: 0, izin: 0, alpa: 0, haid: 0, tidakHadir: 0, persen: 0 }; } function persenTambahStatus_( target, status ) { const s = String( status || "" ) .trim() .toUpperCase(); target.kesempatan++; if ( s === "HAID" ) { target.haid++; return; } /* * Semua selain HAID masuk penyebut. */ target.dihitung++; if ( s === "HADIR" ) { target.hadir++; } else if ( s === "TERLAMBAT" ) { target.terlambat++; } else if ( s === "SAKIT" ) { target.sakit++; } else if ( s === "IZIN" ) { target.izin++; } else if ( s === "ALPA" ) { target.alpa++; } else { /* * Tidak punya baris PRESENSI setelah sesi selesai, * BELUM ABSEN, TIDAK HADIR, atau status lain = tidak hadir. */ target.tidakHadir++; } } function persenFinal_(obj) { obj.persen = persenNilai_( obj.hadir + obj.terlambat, obj.dihitung ); return obj; } /* ===================================================== API UTAMA ===================================================== */ function persentaseKehadiran( payload, token ) { const session = cekSession( token ); if (!session) { return { berhasil: false, harusLogin: true, pesan: "Sesi login habis." }; } payload = payload || {}; let mulai = persenTanggalIso_( payload.tanggalMulai ); let akhir = persenTanggalIso_( payload.tanggalAkhir ); const hariIni = Utilities.formatDate( new Date(), TIMEZONE, "yyyy-MM-dd" ); /* * Default aman: bulan berjalan sampai hari ini. */ if (!akhir) { akhir = hariIni; } if (!mulai) { const dHariIni = persenDateDariIso_( hariIni ); mulai = Utilities.formatDate( new Date( dHariIni.getFullYear(), dHariIni.getMonth(), 1, 12, 0, 0 ), TIMEZONE, "yyyy-MM-dd" ); } if ( mulai > akhir ) { return { berhasil: false, pesan: "Tanggal mulai tidak boleh setelah tanggal akhir." };
